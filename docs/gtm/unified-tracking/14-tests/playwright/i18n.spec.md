# Tests i18n (fr-MA / ar-MA + RTL)

## Locale fr-MA (défaut)

```typescript
test.describe('i18n fr-MA', () => {
  test.use({ locale: 'fr-MA' })

  test('renders all wizard labels in French', async ({ page }) => {
    await page.goto('/admin/tracking/new')
    
    await expect(page.getByRole('heading', { name: 'Choisir les outils' })).toBeVisible()
    await expect(page.getByRole('button', { name: 'Continuer' })).toBeVisible()
    await expect(page.getByText('auto-rempli')).toHaveCount(0)  // pas encore
  })

  test('uses French number/date formats', async ({ page }) => {
    await page.goto('/admin/tracking/plan/active-v8')
    
    // Date format DD/MM/YYYY
    await expect(page.getByText(/12\/05\/2026/)).toBeVisible()
    
    // Espaces insécables avant : ; ? !
    await expect(page.getByText(/Statut\u00a0:/)).toBeVisible()
  })

  test('uses Maroccan French wording (Amal voice)', async ({ page }) => {
    await page.goto('/admin/tracking/new')
    
    // "Mes outils" plutôt que "Sélectionnez vos outils"
    await expect(page.getByText('Mes outils de mesure')).toBeVisible()
  })
})
```

## Locale ar-MA (RTL)

```typescript
test.describe('i18n ar-MA RTL', () => {
  test.use({ locale: 'ar-MA' })

  test('html dir is rtl', async ({ page }) => {
    await page.goto('/admin/tracking/new')
    const html = page.locator('html')
    await expect(html).toHaveAttribute('dir', 'rtl')
    await expect(html).toHaveAttribute('lang', 'ar-MA')
  })

  test('renders headings in Arabic', async ({ page }) => {
    await page.goto('/admin/tracking/new')
    await expect(page.getByRole('heading', { name: 'اختر الأدوات' })).toBeVisible()
  })

  test('stepper progresses right to left', async ({ page }) => {
    await page.goto('/admin/tracking/new')
    
    const step1 = page.getByRole('button', { name: /الخطوة 1/ })
    const step2 = page.getByRole('button', { name: /الخطوة 2/ })
    
    const step1Box = await step1.boundingBox()
    const step2Box = await step2.boundingBox()
    
    // In RTL, step 2 is to the LEFT of step 1
    expect(step2Box!.x).toBeLessThan(step1Box!.x)
  })

  test('form fields show Arabic labels with proper alignment', async ({ page }) => {
    await page.goto('/admin/tracking/new?step=2')
    
    const label = page.getByText('معرف GA4')
    await expect(label).toBeVisible()
    await expect(label).toHaveCSS('text-align', 'right')
  })

  test('icons that have direction flip in RTL (chevron, arrow)', async ({ page }) => {
    await page.goto('/admin/tracking/new')
    
    const chevron = page.locator('[data-testid="back-chevron"]').first()
    // Chevron pointing left in LTR → right in RTL via CSS transform
    const transform = await chevron.evaluate(el => getComputedStyle(el).transform)
    expect(transform).toContain('matrix(-1, 0, 0, 1')  // mirrored
  })
})
```

## Switch de locale

```typescript
test('switching locale persists across pages', async ({ page }) => {
  await page.goto('/admin/tracking')
  
  // Switcher en bas
  await page.getByRole('button', { name: /Langue/ }).click()
  await page.getByRole('menuitem', { name: 'العربية' }).click()
  
  await page.waitForURL(/locale=ar-MA/)
  await expect(page.getByRole('heading', { name: /خطة التتبع/ })).toBeVisible()
  
  // Navigate to another page
  await page.goto('/admin/tracking/history')
  await expect(page.locator('html')).toHaveAttribute('dir', 'rtl')
})
```

## Mélange LTR dans contenu RTL (codes, IDs)

```typescript
test('measurement IDs stay LTR inside RTL layout', async ({ page }) => {
  await page.goto('/admin/tracking/new?step=2&locale=ar-MA')
  
  const idInput = page.getByLabel('معرف GA4')
  await idInput.fill('G-5VHP17SDZM')
  
  // L'input garde la direction LTR pour les codes
  await expect(idInput).toHaveAttribute('dir', 'ltr')
})

test('JSON preview is always LTR', async ({ page }) => {
  await page.goto('/admin/tracking/plan/draft-001?mode=expert&locale=ar-MA')
  
  const preview = page.locator('[data-testid="json-preview"]')
  await expect(preview).toHaveCSS('direction', 'ltr')
})
```
