# Tests accessibilité (axe-playwright)

## Setup

```typescript
import AxeBuilder from '@axe-core/playwright'
import { test, expect } from '@playwright/test'

async function expectNoA11yViolations(page, options = {}) {
  const results = await new AxeBuilder({ page })
    .withTags(['wcag2a', 'wcag2aa', 'wcag21aa'])
    .options(options)
    .analyze()
  
  if (results.violations.length > 0) {
    console.log('A11y violations:', JSON.stringify(results.violations, null, 2))
  }
  expect(results.violations).toEqual([])
}
```

## Modales et états transitoires

```typescript
test.describe('a11y — modales', () => {
  test('confirm activation modale traps focus and labels', async ({ page }) => {
    await page.goto('/admin/tracking/plan/draft-001?step=5')
    await page.getByRole('button', { name: /Activer/ }).click()
    
    const dialog = page.getByRole('dialog')
    await expect(dialog).toBeVisible()
    await expect(dialog).toHaveAttribute('aria-labelledby', /modal-title/)
    
    await expectNoA11yViolations(page)
    
    // Focus dans la modale
    const activeEl = await page.evaluate(() => document.activeElement?.tagName)
    expect(activeEl).toMatch(/BUTTON|H[1-3]/)
  })

  test('toast notifications announce via aria-live', async ({ page }) => {
    await page.goto('/admin/tracking/new')
    await page.getByRole('checkbox', { name: 'GA4' }).check()
    await page.keyboard.press('Meta+S')
    
    const toast = page.getByRole('status').or(page.locator('[aria-live="polite"]'))
    await expect(toast).toContainText(/Sauvegardé/)
  })
})
```

## Lecteurs d'écran (smoke checks)

```typescript
test.describe('a11y — semantic structure', () => {
  test('every page has exactly one h1', async ({ page }) => {
    await page.goto('/admin/tracking')
    const h1Count = await page.locator('h1').count()
    expect(h1Count).toBe(1)
  })

  test('all form fields have associated labels', async ({ page }) => {
    await page.goto('/admin/tracking/new?step=2')
    
    const inputs = await page.locator('input:not([type=hidden])').all()
    for (const input of inputs) {
      const id = await input.getAttribute('id')
      const ariaLabel = await input.getAttribute('aria-label')
      const ariaLabelledBy = await input.getAttribute('aria-labelledby')
      
      if (id) {
        const label = page.locator(`label[for="${id}"]`)
        const hasLabel = await label.count() > 0
        expect(hasLabel || ariaLabel || ariaLabelledBy).toBeTruthy()
      } else {
        expect(ariaLabel || ariaLabelledBy).toBeTruthy()
      }
    }
  })

  test('landmark structure: header, main, nav, footer', async ({ page }) => {
    await page.goto('/admin/tracking')
    
    await expect(page.locator('header[role=banner], header').first()).toBeVisible()
    await expect(page.locator('main[role=main], main').first()).toBeVisible()
    await expect(page.locator('nav').first()).toBeVisible()
  })

  test('skip-to-content link is first focusable', async ({ page }) => {
    await page.goto('/admin/tracking')
    await page.keyboard.press('Tab')
    
    const focused = page.locator(':focus')
    await expect(focused).toContainText(/Aller au contenu/i)
  })
})
```

## Animation reduce

```typescript
test('respects prefers-reduced-motion', async ({ page, context }) => {
  await context.emulateMedia({ reducedMotion: 'reduce' })
  await page.goto('/admin/tracking/new')
  
  await page.getByRole('checkbox', { name: 'GA4' }).check()
  await page.getByRole('button', { name: /Continuer/ }).click()
  
  // L'animation entre steps doit être désactivée
  const stepContainer = page.locator('[data-testid="step-container"]')
  const transition = await stepContainer.evaluate(el => getComputedStyle(el).transition)
  
  expect(transition).toMatch(/none|0s/)
})
```
