QUERIES TYPES POST-REFONTE

-- KPIs 24h par provider
SELECT
  kind,
  count(*) as total,
  count(*) FILTER (WHERE (providers_results->kind->>'status') = 'success') as ok,
  avg((providers_results->kind->>'latencyMs')::int) as avg_latency
FROM tracking_events_log,
     unnest(providers_dispatched) as kind
WHERE received_at >= now() - interval '24 hours'
GROUP BY kind;

-- Détecter les doublons par event_id (vérif dedup serveur)
SELECT event_id, count(*) FROM tracking_events_log
WHERE event_id IS NOT NULL
GROUP BY event_id HAVING count(*) > 1;
