# Test ultime M5.7 — Template editor

> File: `apps/web/e2e/m5.7-ultimate.spec.ts`

## Scénario

> Un admin crée un nouveau template depuis le "default FemiGlow",
> modifie le titre, ajoute un CTA secondaire, preview avec un lead
> réel, envoie un test, rollback à v1, vérifie rollback OK.

```typescript
test('M5.7 — Template editor full flow', async ({ page, request }) => {
  test.setTimeout(120_000);

  // === 1. Open templates list ===
  await page.goto('/admin/emails/templates');
  await expect(page.getByRole('heading', { name: /templates/i })).toBeVisible();

  // === 2. Create new from default ===
  await page.click('button:has-text("Nouveau template")');
  await page.click('text=Default FemiGlow');
  
  await page.fill('input[name="name"]', 'Test ultimate template');
  // slug auto generated
  await page.click('button:has-text("Créer")');

  // === 3. Editor opens with default content ===
  await expect(page.locator('[data-testid="template-editor"]')).toBeVisible();

  // === 4. Modify title via Sections view ===
  await page.click('text=Titre');
  await page.fill('[data-testid="section-title-input"]', 'Bonjour {{firstName}}, voici une promo');
  await page.click('button:has-text("OK")');

  // === 5. Add secondary CTA from library ===
  await page.click('button:has-text("Insérer un composant")');
  await page.click('text=CTA secondaire ghost');
  // Drop it after the primary CTA
  
  await page.fill('[data-testid="cta-label-input"]', 'En savoir plus');
  await page.fill('[data-testid="cta-url-input"]', 'https://femiglow-maroc.com/rituel');
  
  // === 6. Preview live ===
  const previewFrame = page.frameLocator('iframe[data-testid="preview-iframe"]');
  await expect(previewFrame.locator('text=voici une promo')).toBeVisible();
  await expect(previewFrame.locator('text=En savoir plus')).toBeVisible();

  // === 7. Switch to real lead context ===
  await page.click('[data-testid="variables-source"]');
  await page.click('text=Vrai contact');
  await page.fill('[data-testid="real-lead-search"]', 'fatima');
  await page.click('text=fatima@example.com');
  
  // Variables panel updates
  await expect(page.locator('text=firstName: "Fatima"')).toBeVisible();
  await expect(page.locator('text=city: "Casablanca"')).toBeVisible();

  // === 8. Save (v2) ===
  await page.keyboard.press('Meta+S');
  await page.fill('[data-testid="commit-message"]', 'Add secondary CTA');
  await page.click('button:has-text("Enregistrer")');
  
  await expect(page.locator('text=Sauvegardé v2')).toBeVisible();

  // === 9. Test send ===
  await page.click('button:has-text("Test send")');
  // Modal opens
  await page.click('text=Mock 1');
  await page.click('button:has-text("Envoyer le test")');
  
  await expect(page.locator('text=Test envoyé')).toBeVisible({ timeout: 10_000 });
  
  // Verify outbox row created via API
  const outboxRes = await request.get('/api/admin/emails/transactional/search', {
    data: { 
      filters: { source: 'admin.template-test' },
      pagination: { limit: 10, offset: 0 },
    },
  });
  expect((await outboxRes.json()).rows.length).toBeGreaterThan(0);

  // === 10. Check version history ===
  await page.click('button:has-text("Historique")');
  await expect(page.locator('text=v2 (actuel)')).toBeVisible();
  await expect(page.locator('text=v1')).toBeVisible();
  await expect(page.locator('text=Add secondary CTA')).toBeVisible();

  // === 11. Rollback to v1 ===
  await page.click('[data-testid="rollback-v1"]');
  // Confirmation modal
  await page.click('button:has-text("Rollback")');
  
  await expect(page.locator('text=Rollback effectué')).toBeVisible();
  
  // === 12. Verify template state matches v1 ===
  // The secondary CTA should be gone, the title back to default
  await page.click('text=Éditeur');
  await expect(previewFrame.locator('text=En savoir plus')).not.toBeVisible();
});

test('M5.7 — Sanitization rejects script', async ({ page }) => {
  await page.goto('/admin/emails/templates/test-ultimate-template/edit');
  
  // Switch to source view
  await page.click('text=Source');
  
  // Insert malicious script
  await page.click('[data-testid="monaco-editor"]');
  await page.keyboard.press('Meta+End');
  await page.keyboard.type('<script>alert("XSS")</script>');
  
  // Try to save
  await page.keyboard.press('Meta+S');
  
  // Should warn + sanitize
  await expect(page.locator('text=script non autorisé')).toBeVisible();
});

test('M5.7 — Variables panel shows unresolved warnings', async ({ page }) => {
  await page.goto('/admin/emails/templates/test-ultimate-template/edit');
  
  await page.click('text=Source');
  await page.keyboard.press('Meta+End');
  await page.keyboard.type('<p>{{unknownVar}}</p>');
  
  // Variables panel should show warning
  await expect(page.locator('[data-testid="variable-warning"]:has-text("unknownVar")')).toBeVisible();
});
```
