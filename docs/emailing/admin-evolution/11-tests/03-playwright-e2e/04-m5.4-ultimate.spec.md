# Test ultime M5.4 — Campaigns avec audiences natives

> File: `apps/web/e2e/m5.4-ultimate.spec.ts`

## Scénario

> Créer audience de 50 emails de test → créer campagne → finalize →
> vérifier que Listmonk a reçu la liste éphémère + a envoyé.

```typescript
test('M5.4 — Campaign send to native audience', async ({ page, request }) => {
  test.setTimeout(120_000);
  
  // Pre : 50 leads de test en DB (seed)
  
  // === 1. Create audience targeting these 50 leads ===
  await page.goto('/admin/emails/audiences/new');
  await page.fill('input[name="name"]', `Test campaign ${Date.now()}`);
  await page.click('button:has-text("Continuer")');
  
  // Rule : email contains 'test-m5-4'
  await page.click('button:has-text("Ajouter un critère")');
  await page.click('text=Email');
  await page.selectOption('select[name="operator"]', 'contains');
  await page.fill('input[name="value"]', 'test-m5-4');
  
  await page.waitForTimeout(1_000);
  await expect(page.locator('text=50 contact')).toBeVisible();
  
  await page.click('button:has-text("Continuer")');
  await page.click('button:has-text("Créer l\'audience")');
  
  // === 2. Create campaign ===
  await page.goto('/admin/emails/campaigns/new');
  // ... fill name, audience (select the one we just created), template, etc.
  
  // === 3. Finalize ===
  await page.click('button:has-text("Lancer l'envoi")');
  
  // Wait for finalize complete
  await expect(page.locator('text=Campagne envoyée')).toBeVisible({ timeout: 30_000 });
  
  // === 4. Verify Listmonk has the list ===
  const listmonkLists = await request.get('/api/admin/emails/__test__/listmonk-lists');
  const lists = await listmonkLists.json();
  const ephemeralList = lists.find((l: any) => l.name.startsWith('fg-'));
  expect(ephemeralList).toBeDefined();
  expect(ephemeralList.subscriber_count).toBe(50);
  
  // === 5. Verify Listmonk has the campaign ===
  const lmCampaigns = await request.get('/api/admin/emails/__test__/listmonk-campaigns');
  const campaigns = await lmCampaigns.json();
  const recent = campaigns.find((c: any) => c.list_ids.includes(ephemeralList.id));
  expect(recent).toBeDefined();
  expect(recent.status).toMatch(/scheduled|running|finished/);
});
```
