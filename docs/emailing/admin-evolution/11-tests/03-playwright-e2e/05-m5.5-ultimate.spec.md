# Test ultime M5.5 — Automation studio

> File: `apps/web/e2e/m5.5-ultimate.spec.ts`

## Scénario

> Un admin crée une automation "post-purchase upsell" complète avec
> branch, déclenche une commande de test, et vérifie que la branche YES
> et NO se déclenchent correctement.

```typescript
test('M5.5 — Automation post-purchase with branch', async ({ page, request }) => {
  test.setTimeout(180_000);  // 3min
  
  // === Create automation ===
  await page.goto('/admin/emails/automation/new');
  
  // Step 1
  await page.fill('input[name="name"]', 'Post-purchase upsell test');
  await page.click('button:has-text("Continuer")');
  
  // Step 2 — Trigger
  await page.click('text=Événement');
  await page.click('[data-testid="event-picker"]');
  await page.fill('[data-testid="event-search"]', 'order');
  await page.click('text=order.placed');
  await page.click('button:has-text("Continuer")');
  
  // Step 3 — Steps
  // 1. Wait 5s (test value)
  await page.click('button:has-text("Ajouter une étape")');
  await page.click('text=Wait');
  await page.fill('input[name="duration"]', '5s');
  await page.click('button:has-text("Enregistrer")');
  
  // 2. Send welcome
  await page.click('button:has-text("Ajouter une étape")');
  await page.click('text=Send');
  await page.selectOption('select[name="template"]', 'welcome');
  await page.click('button:has-text("Enregistrer")');
  
  // 3. Branch
  await page.click('button:has-text("Ajouter une étape")');
  await page.click('text=Branch');
  // ... configure condition "user opened step 2"
  await page.click('button:has-text("Enregistrer")');
  
  // 3a (yes branch) - Send upsell
  await page.click('[data-testid="branch-yes-add"]');
  await page.click('text=Send');
  await page.selectOption('select[name="template"]', 'cart-abandoned');  // use any template
  await page.click('button:has-text("Enregistrer")');
  
  // 3a (no branch) - Tag cart_lost
  await page.click('[data-testid="branch-no-add"]');
  await page.click('text=Tag');
  await page.fill('input[name="tag"]', 'no_upsell');
  await page.click('button:has-text("Enregistrer")');
  
  await page.click('button:has-text("Continuer")');
  
  // Step 4 — Frequency
  await page.click('button:has-text("Continuer")');
  
  // Step 5 — Activate
  await page.click('text=Activer immédiatement');
  await page.click('button:has-text("Créer + activer")');
  
  // === Trigger by API (simulate order.placed) ===
  await request.post('/api/__test__/trigger-event', {
    data: { 
      event_name: 'order.placed', 
      email: 'test-upsell@example.com',
      properties: { total: 500 },
    },
  });
  
  // === Wait for cron tick (5s wait + ~30s buffer) ===
  await page.waitForTimeout(45_000);
  
  // === Check run state ===
  await page.goto('/admin/emails/automation');
  await page.click('text=Post-purchase upsell test');
  
  await page.click('text=test-upsell@example.com');
  await expect(page.locator('text=Step 1 done')).toBeVisible();
  await expect(page.locator('text=Step 2 done')).toBeVisible();
  await expect(page.locator('text=Step 3 — waiting')).toBeVisible();  // branch waiting on opened
  
  // === Simulate "opened" event ===
  await request.post('/api/__test__/simulate-event', {
    data: { email: 'test-upsell@example.com', event_name: 'email.opened' },
  });
  
  await page.waitForTimeout(35_000);
  await page.reload();
  await expect(page.locator('text=Step 3.yes advanced')).toBeVisible();
  await expect(page.locator('text=Send upsell email')).toBeVisible();
});
```
