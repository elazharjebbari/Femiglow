# Test ultime M5.3 — Audience builder

> File: `apps/web/e2e/m5.3-ultimate.spec.ts`

## Scénario

> Un admin crée une audience "Cart abandoners 7d" via le wizard, voit
> le preview correct (count + sample), snapshot manuellement.

```typescript
test('M5.3 — Audience wizard end-to-end', async ({ page }) => {
  test.setTimeout(120_000);
  
  // Pré : seed user_event avec cart.abandoned pour ~50 users dans last 7d
  
  // === Open list page ===
  await page.goto('/admin/emails/audiences');
  await expect(page.getByRole('heading', { name: /audiences/i })).toBeVisible();
  
  // === Create new ===
  await page.click('button:has-text("Nouvelle audience")');
  
  // === Step 1 — Métadonnées ===
  await page.fill('input[name="name"]', 'Cart abandoners 7d');
  await expect(page.locator('input[name="slug"]')).toHaveValue('cart-abandoners-7d');
  await page.click('button:has-text("Continuer")');
  
  // === Step 2 — Rules ===
  await page.click('button:has-text("Ajouter un critère")');
  await page.click('text=Événement utilisateur');
  await page.selectOption('select[name="event"]', 'cart.abandoned');
  await page.fill('input[name="within"]', '7d');
  
  // Preview live (debounce 800ms)
  await page.waitForTimeout(1_000);
  await expect(page.locator('[data-testid="audience-preview-count"]')).toHaveText(/\d+ contacts/);
  const count = await page.locator('[data-testid="audience-preview-count"]').textContent();
  expect(parseInt(count!.match(/\d+/)![0])).toBeGreaterThan(0);
  
  // Sample
  await page.click('button:has-text("Voir 10 exemples")');
  await expect(page.locator('[data-testid="audience-sample"] tr')).toHaveCount(10);
  
  await page.click('button:has-text("Continuer")');
  
  // === Step 3 — Recap + create ===
  await expect(page.locator('text=Cart abandoners 7d')).toBeVisible();
  await page.click('button:has-text("Créer l\'audience")');
  
  // Detail page
  await expect(page).toHaveURL(/\/audiences\/[a-f0-9-]+$/);
  
  // === Snapshot ===
  await page.click('button:has-text("Snapshot maintenant")');
  
  // Loading toast
  await expect(page.locator('text=Snapshot en cours')).toBeVisible();
  
  // Done within 30s
  await expect(page.locator('text=Snapshot terminé')).toBeVisible({ timeout: 30_000 });
  
  // Snapshot row visible
  await expect(page.locator('[data-testid="snapshot-row"]').first()).toBeVisible();
});

test('M5.3 — Empty audience returns 0', async ({ page }) => {
  await page.goto('/admin/emails/audiences/new');
  await page.fill('input[name="name"]', 'Empty test');
  await page.click('button:has-text("Continuer")');
  
  // Rule qui ne matche personne
  await page.click('button:has-text("Ajouter un critère")');
  await page.click('text=Nombre de commandes');
  await page.selectOption('select[name="operator"]', 'gte');
  await page.fill('input[name="value"]', '999999');
  
  await page.waitForTimeout(1_000);
  await expect(page.locator('text=0 contact')).toBeVisible();
});
```
