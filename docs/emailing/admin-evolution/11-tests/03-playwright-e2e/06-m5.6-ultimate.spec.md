# Test ultime M5.6 — Keyboard-only journey

> File: `apps/web/e2e/m5.6-ultimate.spec.ts`

## Scénario

> Un admin réalise tout le parcours critique au clavier UNIQUEMENT
> (no mouse).

```typescript
test('M5.6 — Full keyboard-only journey', async ({ page }) => {
  test.setTimeout(60_000);
  
  // === Login is via storage state, skip ===
  await page.goto('/admin/emails');
  
  // === Navigate to transactional via shortcut ===
  await page.keyboard.press('g');
  await page.keyboard.press('t');
  await expect(page).toHaveURL(/\/admin\/emails\/transactional/);
  
  // === Cmd-K search ===
  await page.keyboard.press('Meta+K');
  await page.keyboard.type('status:failed');
  await page.keyboard.press('Enter');
  
  // === Navigate down to a row ===
  await page.keyboard.press('j');
  await page.keyboard.press('j');
  
  // === Open detail ===
  await page.keyboard.press('Enter');
  await expect(page.url()).toMatch(/\/transactional\/[a-f0-9-]+/);
  
  // === Retry via shortcut ===
  await page.keyboard.press('r');
  // ... confirmation modal opens, focus is on cancel by default
  await page.keyboard.press('Tab');  // focus on confirm
  await page.keyboard.press('Enter');
  await expect(page.locator('text=retried')).toBeVisible();
  
  // === Navigate to audiences ===
  await page.keyboard.press('g');
  await page.keyboard.press('a');
  await expect(page).toHaveURL(/\/audiences/);
  
  // === Create new (shortcut) ===
  await page.keyboard.press('n');
  await expect(page).toHaveURL(/\/audiences\/new/);
  
  // === Fill wizard via keyboard ===
  await page.keyboard.type('Test keyboard audience');
  await page.keyboard.press('Tab');  // navigate to next field
  // ...
});

test('M5.6 — Cheat sheet via ? key', async ({ page }) => {
  await page.goto('/admin/emails');
  await page.keyboard.press('?');
  await expect(page.locator('[role="dialog"]')).toBeVisible();
  await expect(page.locator('text=Raccourcis clavier')).toBeVisible();
  await page.keyboard.press('Escape');
  await expect(page.locator('[role="dialog"]')).not.toBeVisible();
});

test('M5.6 — A11y axe scan on all main pages', async ({ page }) => {
  const pages = [
    '/admin/emails',
    '/admin/emails/transactional',
    '/admin/emails/audiences',
    '/admin/emails/audiences/new',
    '/admin/emails/campaigns',
    '/admin/emails/automation',
    '/admin/emails/automation/new',
  ];
  
  for (const url of pages) {
    await page.goto(url);
    await injectAxe(page);
    const violations = await getViolations(page);
    const critical = violations.filter(v => v.impact === 'critical');
    expect(critical, `${url} has critical a11y violations`).toEqual([]);
  }
});

test('M5.6 — Reduced motion respected', async ({ page }) => {
  await page.emulateMedia({ reducedMotion: 'reduce' });
  await page.goto('/admin/emails/transactional');
  
  // Vérifier que les transitions sont à 0ms
  const transition = await page.locator('[data-testid="kpi-card"]').first()
    .evaluate(el => getComputedStyle(el).transitionDuration);
  expect(transition).toBe('0.01ms');
});
```
